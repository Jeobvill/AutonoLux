SCROLL_ARRIBA MACRO LINEAS,REN_SUP_IZQ,COL_SUP_IZQ,REN_INF_DER,COL_INF_DER,COLOR
	MOV AH,6
	MOV AL,LINEAS
	MOV CH,REN_SUP_IZQ
	MOV CL,COL_SUP_IZQ
	MOV DH,REN_INF_DER
	MOV DL,COL_INF_DER
	MOV BH,COLOR
	INT 10H
SCROLL_ARRIBA ENDM
;======================================== 4 NOV
    CURSOR MACRO REN,COL,PAG
        MOV AH,2
        MOV DH,REN
        MOV DL,COL
        MOV BH,PAG
        INT 10H
    CURSOR ENDM
 
    IMP_SINCOLOR MACRO CADENA
        MOV AH,9
        LEA DX,CADENA
        INT 21H
    IMP_SINCOLOR ENDM
 
    RASTREO MACRO
        MOV AH,0
        INT 16H
    RASTREO ENDM
 
    IMPRIMIR MACRO DATO
        MOV AH,5
        MOV DL,DATO
        INT 21H
    IMPRIMIR ENDM

    NEWPAGE MACRO
        MOV AH,5
        MOV DL,12
        INT 21H
    NEWPAGE ENDM
 
    ;====MACROS PARA CARPETAS===
    CREAR_CARPETA MACRO RUTA
        MOV AH,39H
        LEA DX,RUTA
        INT 21H
    CREAR_CARPETA ENDM
 
    ;====MACROS PARA ARCHIVOS===
    CREAR_ARCHIVO MACRO RUTA,TIPO
        ;NOTA: El tipo de archivo 32 es archivo de USUARIO
        MOV AH,3CH
        LEA DX,RUTA
        MOV CX,TIPO
        INT 21H
    CREAR_ARCHIVO ENDM

    ABRIR_ARCHIVO MACRO RUTA,MODO
       ;MODO
       ;0 SOLO LECTURA
       ;1 SOLO ESCRITURA
       ;2 LECTURA Y ESCRITURA
       MOV AH,3DH
       LEA DX,RUTA
       MOV AL,MODO
       INT 21H
   	  	MOV ID,AX
    ABRIR_ARCHIVO ENDM

    ESCRIBIR_ARCHIVO MACRO ID,CANTIDAD,DATOS
       MOV AH,40H
       MOV BX,ID
       MOV CX,CANTIDAD
       LEA DX,DATOS
       INT 21H
    ESCRIBIR_ARCHIVO ENDM

    LEER_ARCHIVO MACRO CUANTOS,BUFFER_LEIDOS,ID
    	MOV AH,3FH
    	LEA DX,BUFFER_LEIDOS
    	MOV CX,CUANTOS
	MOV BX,ID
    	INT 21H
    LEER_ARCHIVO ENDM

    LECTURA_CADENA MACRO CADENA
    	MOV AH,10
        LEA DX,CADENA
        INT 21H
    LECTURA_CADENA ENDM

    CERRAR_ARCHIVO MACRO ID
	MOV AH,3EH
	MOV BX,ID
	INT 21H
    CERRAR_ARCHIVO ENDM

    ESCRIBIR_CARCOLOR MACRO VECES,CARACTER,COLOR
		MOV AH,9
		MOV CX,VECES
		MOV AL,CARACTER
		MOV BL,COLOR
		INT 10H
    ESCRIBIR_CARCOLOR ENDM

    IMPCAD_COLOR MACRO CADENA,LONG,REN,COL,PAG,MODO,COLOR
		MOV AH,19
    		LEA BP,CADENA
    		MOV CX,LONG
    		MOV DH,REN
    		MOV DL,COL
    		MOV BH,PAG
    		MOV AL,MODO
    		MOV BL,COLOR
    		INT 10H
	
    IMPCAD_COLOR ENDM
